// Procedures and control flow
proc foo(n: Int) -> (int, int): // Tuple
	var
		x: Int = 10;
		y: Int; // Zero initialized is default

		tup: My_Tuple;
begin
	y = match x with
		value -> value,
		error -> error,
	end;

	// variant checking
	// expr `is` variant_label (`as` alt_name)?
	if x is error as e do
		return e;
	end

	// shorthands that work for Option[T] and Result[T, E]
	y = x or_return;
	y = x or_else 0;

	for n > 0 do
		x = x * 2;
		y = y + 1;
	end

end
