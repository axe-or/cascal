proc safe_div(a: Int, b: Int) -> (Int, Bool) {
	if b == 0 {
		return 0, false;
	}
	return a / b, true;
}

proc main(){
	var q, ok = safe_div(30, 0);
	assert(ok, "Divide by zero");
}
